const RAW_MAZE = "00100|00101|10000|01110|00010"; const DIR_ROWS = [-1, 1, 0, 0]; const DIR_COLS = [0, 0, -1, 1]; let chartInstance = null; function pseudoRandom(s) { let mask = 4294967295; let w = (123456789 + s) & mask; let z = (987654321 - s) & mask; return function() { z = (36969 * (z & 65535) + (z >> 16)) & mask; w = (18000 * (w & 65535) + (w >> 16)) & mask; return (((z << 16) + w) >> 16) + 32768 / 65536; } } function calculateImprint(r, c, maze) { let touch = 0; for(let i = 0; i < 4; i++) { let nr = r + DIR_ROWS[i]; let nc = c + DIR_COLS[i]; if (nr < 0 || nr >= 5 || nc < 0 || nc >= 5) { touch++; } else if (maze && maze[nr] && maze[nr][nc] === 1) { touch++; } } let dist = Math.sqrt(Math.pow(r - 4, 2) + Math.pow(c - 4, 2)); let sound = Math.round((dist / (dist + 1)) * 10) / 10; return touch + "_" + sound + "_" + ((r + c) % 3); } function runCoreEngine(useThermo, startEnergy, simCost, actCost) { let nextRand = pseudoRandom(105); let maze = RAW_MAZE.split("|").map(row => row.split("").map(Number)); let r = 0, c = 0; let energy = startEnergy; let history = []; history.push(Number(startEnergy)); let steps = 0, vetoes = 0, targetReached = false; let memoryTraces = {}; for(let i = 1; i < 100; i++) { if((r === 4 && c === 4) || energy <= 0) { if(r === 4 && c === 4) targetReached = true; break; } if(i % 15 === 0) { let mr = Math.floor(nextRand() * 3) + 1; let mc = Math.floor(nextRand() * 3) + 1; if (mr < 5 && mc < 5 && maze[mr]) maze[mr][mc] = maze[mr][mc] === 0 ? 1 : 0; } let dirIndices =; for (let j = dirIndices.length - 1; j > 0; j--) { let k = Math.floor(nextRand() * (j + 1)); let temp = dirIndices[j]; dirIndices[j] = dirIndices[k]; dirIndices[k] = temp; } if(useThermo) { let bestPath = null; let minTrace = 99999; for(let idx = 0; idx < 4; idx++) { let didx = dirIndices[idx]; let nr = r + DIR_ROWS[didx]; let nc = c + DIR_COLS[didx]; if(nr < 0 || nr >= 5 || nc < 0 || nc >= 5 || (maze[nr] && maze[nr][nc] === 1)) continue; let token = calculateImprint(nr, nc, maze); let traceCount = memoryTraces[token] || 0; if(traceCount < minTrace) { minTrace = traceCount; bestPath = { nr: nr, nc: nc, token: token, trace: traceCount }; } } if(bestPath !== null) { if(bestPath.trace > 1) { vetoes++; energy -= simCost; history.push(Number(energy)); continue; } else { energy -= simCost; energy -= actCost; r = bestPath.nr; c = bestPath.nc; memoryTraces[bestPath.token] = (memoryTraces[bestPath.token] || 0) + 1; steps++; } } else { vetoes++; energy -= simCost; history.push(Number(energy)); continue; } } else { let dirIdx = Math.floor(nextRand() * 4); let nr = r + DIR_ROWS[dirIdx]; let nc = c + DIR_COLS[dirIdx]; energy -= actCost; if(nr >= 0 && nr < 5 && nc >= 0 && nc < 5 && maze[nr] && maze[nr][nc] !== 1) { r = nr; c = nc; steps++; } } history.push(Number(energy)); } return { history, success: targetReached, steps, vetoes }; } function executeAGISimulation() { let energy = parseInt(document.getElementById('param_energy').value); let costSim = parseInt(document.getElementById('param_sim').value); let costAct = parseInt(document.getElementById('param_act').value); let blind = runCoreEngine(false, energy, costSim, costAct); let thermo = runCoreEngine(true, energy, costSim, costAct); document.getElementById('m_b_res').innerText = blind.success ?
